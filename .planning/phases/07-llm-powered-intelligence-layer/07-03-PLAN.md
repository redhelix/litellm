<interfaces>
useIntelligence hook contract (from research §Pattern 6, mirrors useRequestLog):
```typescript
export interface IntelligenceResult {
  generated_at: string | null
  model_used: string | null
  health_summary: string | null
  anomalies: Array<{ title: string; severity: 'low' | 'medium' | 'high'; description: string }>
  recommendations: Array<{ title: string; body: string }>
  hf_models: Array<{ model_id: string; hf_url: string; tags: string[]; likes: number; downloads?: number; last_modified: string }>
}

export function useIntelligence(sidecarUrl: string): {
  data: IntelligenceResult | null
  loading: boolean
  error: string | null
}
```

```typescript
const res = await fetch(`${sidecarUrl}/api/intelligence/query`, {
```
